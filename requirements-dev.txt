# The pack's own test venv (scripts/verify.sh --fast), NOT what any
# container runs -- apps/*/requirements.txt are the runtime pins, frozen
# from the images with hashes. Those hashes resolve to linux wheels
# (pydantic-core is platform-specific), so they cannot double as the dev
# venv spec on a laptop. Versions here match what the images ship, minus
# the hashes, so a test sees the same libraries the app does.
#
# Kept in sync by hand with apps/join-api/requirements.txt (fastapi's
# version there, 0.141.1, is the newer of the two apps' pins -- console
# stays on 0.140.0). starlette is pinned explicitly rather than left to
# float in as fastapi's dependency (fastapi only requires
# starlette>=0.46.0): both apps' requirements.txt independently resolved
# to starlette==1.3.1, and apps/join-api/app.py's body-size-limit
# middleware relies on private starlette internals
# (_CachedRequest.wrapped_receive's _body-before-_stream_consumed check)
# that have changed across majors -- an unpinned install here previously
# floated to whatever starlette was newest (1.6.0, as of this fix),
# silently testing that middleware against a starlette the shipped image
# never runs.
pytest==9.1.1
fastapi==0.141.1
starlette==1.3.1
httpx==0.28.1
PyYAML==6.0.3
# Matches apps/join-api/requirements.txt's uvicorn==0.52.1 (minus the
# per-platform hashes -- see this file's own header) -- needed for
# test_app_concurrency.py, which starts a real uvicorn server to catch
# thread-affinity bugs a serial TestClient can't reproduce.
uvicorn==0.52.1
# Matches apps/join-api/requirements.txt's psycopg==3.3.4 (minus the
# per-platform hashes -- see this file's own header) -- needed for
# test_store_postgres.py and the psycopg.importorskip'd masking test in
# test_store.py to actually run instead of silently skipping ("--fast with
# and without KP2_TEST_DB_URL").
psycopg[binary]==3.3.4
